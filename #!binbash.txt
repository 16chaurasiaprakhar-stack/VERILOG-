#!/bin/bash

# Step 1: 
verilator --binary -j 0 -Wall \
    fifo.v fifo_tb.v \
    --top fifo_tb \
    --timing \
    --CFLAGS "-std=c++20" \
    --trace

# Step 2: 
cd obj_dir || {
    echo "obj_dir not found"
    exit 1
}

# Step 3: 
make -f Vfifo_tb.mk Vfifo_tb || {
    echo "Compilation failed"
    exit 1
}

# Step 4:  
./Vfifo_tb || {
    echo "Simulation failed"
    exit 1
}

# Step 5: 
gtkwave dump.vcd
